> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Every number the app shows as a measured fact on a US screen is measured on US data. That covers the risk grade and liquidity floors, the forecast ranges, the P/E and dividend bands, the liquidity tiers and the Market Condition card. EG output stays unchanged.

**Architecture:**
- A new package `app/core/calibration/` holds one frozen `Calibration` bundle per market:
  - `types.py`: the dataclasses.
  - `eg.py`: today's EG numbers, moved verbatim.
  - `us.py`: generated by `python -m scripts.calibrate --market US --write`.
  - `for_market(market)`: raises for a market without a bundle.
- Every consumer reads the bundle of the market it serves. The old module-level constants are deleted, and an AST guard test keeps them out of `app/`.
- `scripts/backtest.py`, `scripts/calibrate.py` and `scripts/backfill_macro.py` gain US support. A controller-run task fits the US bundle against the local DB, then later tasks switch the US surfaces on.

**Tech Stack:**
- Backend: FastAPI, psycopg3, pandas/numpy, pytest, tvDatafeed (vendored egxpy).
- Shared web `lib/`: TypeScript, synced verbatim into mobile.
- Mobile: Expo / React Native, jest.

**Spec:** `egx-api-be/docs/superpowers/specs/2026-10-16-us-calibration-design.md`

`app/core/calibration/types.py`:

```python
"""
Calibration bundles: every number the app states as a MEASURED FACT about a
market, grouped per market so a US screen can never quote an EGX fit.

One frozen `Calibration` per market lives beside this file (eg.py, us.py) and
is read through `app.core.calibration.for_market`. Nothing outside this
package may hold a fitted constant — tests/test_calibration_guard.py fails if
one of the old module-level names comes back.
"""
from __future__ import annotations

import dataclasses
from dataclasses import dataclass
from typing import Mapping, Optional


@dataclass(frozen=True)
class ForecastCal:
    fitted_at: str
    n_observations: int
    universe_label: str
    sigma_window: int
    fit_horizon_days: int
    band_width_pct_of_spot: Mapping[str, float]          # {"median", "p25", "p75"}
    z_cone: Mapping[int, float]                          # {50, 80, 90, 95}
    one_sigma_coverage: Mapping[str, Mapping[str, float]]  # {"ewma"|"trailing": {daily, weekly, monthly}}
    gaussian_cone_90_coverage: float   # what the textbook 90% band delivered
    cone_90_coverage: float            # what the fitted 90% band delivered
    daily_band_note: Optional[str]
    cone_note: str

    def payload(self) -> dict:
        """The `calibration` dict expected_move / outcome_band ship. Shape and
        key order unchanged since 2026-09-02, so clients see no difference."""
        return {
            "fitted_at": self.fitted_at,
            "n_observations": self.n_observations,
            "universe": self.universe_label,
            "sigma_window": self.sigma_window,
            "fit_horizon_days": self.fit_horizon_days,
            "band_width_pct_of_spot_median": self.band_width_pct_of_spot["median"],
            "band_width_pct_of_spot_p25": self.band_width_pct_of_spot["p25"],
            "band_width_pct_of_spot_p75": self.band_width_pct_of_spot["p75"],
        }


@dataclass(frozen=True)
class RegimeBand:
    label: str
    summary: str
    median_3m_pct: float
    positive_rate: float
    observations: int


@dataclass(frozen=True)
class RegimeCal:
    fitted_at: str
    weak_max: float
    mixed_max: float
    rho: float              # overlapping rank correlation, every reading
    t: float                # Newey-West, lag = overlap depth
    n: int
    significant: bool       # |t| >= 1.96
    phase_rhos: tuple       # all de-overlapped phases, never just one
    bands: Mapping[str, RegimeBand]   # "weak" | "mixed" | "broad"
    universe_indexes: tuple # index ids the reading averages over
    universe_label: str     # "EGX30 and EGX70", "S&P 500"
    failure_outcome: str    # the /api/calibration "what failed" outcome line

    @property
    def show(self) -> bool:
        """A wrong-signed association is hidden: a card saying "broad
        strength" that preceded WEAKER markets would mislead. A weak but
        right-signed one ships as context (EG: t=1.74)."""
        return self.rho > 0


@dataclass(frozen=True)
class RiskCal:
    fitted_at: str
    n_observations: int
    n_symbols: int
    n_dates: int
    lookback_days: int
    forward_days: int
    vol_predicts_vol_ic: float
    vol_predicts_vol_t: float
    vol_predicts_vol_ic_nonoverlapping: float
    vol_predicts_vol_t_nonoverlapping: float
    vol_predicts_drawdown_ic: float
    vol_predicts_drawdown_t: float
    vol_predicts_drawdown_ic_nonoverlapping: float
    vol_predicts_drawdown_t_nonoverlapping: float
    quintiles: Mapping[int, Mapping[str, float]]

    def payload(self) -> dict:
        """The `calibration` dict /api/risk ships (every field but quintiles)."""
        return {f.name: getattr(self, f.name)
                for f in dataclasses.fields(self) if f.name != "quintiles"}


@dataclass(frozen=True)
class FundamentalsCal:
    label: str              # the word in "the EGX median" / "the US median"
    pe_median: float
    pe_bands: tuple         # ((upper, points), ...) half-open "< upper"; last upper is math.inf
    dy_median: float
    dy_bands: tuple
    dy_mid_under_risk_free: bool  # the +8 yield band's upper bound sits below the market's cash rate


@dataclass(frozen=True)
class Calibration:
    market: str
    forecast: ForecastCal
    regime: RegimeCal
    risk: RiskCal
    fundamentals: FundamentalsCal
    liquidity_tiers: Mapping[str, tuple]   # {tier: (thin_below, low_below)} shares/day
    liquidity_default_tier: str
    liquid_min_turnover: Optional[float]   # mirrors MarketConfig; a test pins they agree
    liquid_min_price: Optional[float]
    measured_failures: tuple = ()          # /api/calibration "what failed" entries measured on THIS market
    breadth_measured: bool = False         # breadth's evidence numbers were measured on this market
    dividend_drag_measured: bool = False   # the price-vs-total-return drag was measured on this market
    provisional: bool = False              # True only until the market's bundle is fitted


def band_index(bands, value) -> int:
    """Index of the half-open band (`value < upper`) that `value` falls in."""
    for i, (upper, _points) in enumerate(bands):
        if value < upper:
            return i
    return len(bands) - 1
```

```python
def for_market(market) -> Calibration:
    """The bundle for a market code ("EG") or MarketConfig. KeyError if none."""
    code = market if isinstance(market, str) else getattr(market, "code", None)
    key = code.strip().upper() if isinstance(code, str) else None
```

```python
import pathlib

APP = pathlib.Path(__file__).resolve().parents[1] / "app"

FORBIDDEN = {
    "EGX_Z_CONE_60D", "ONE_SIGMA_COVERAGE_PCT", "CALIBRATION",

def offences(names: set) -> list:
    found = []
    for path in sorted(APP.rglob("*.py")):
        if "vendor" in path.parts:
            continue
        rel = path.relative_to(APP)
        tree = ast.parse(path.read_text(encoding="utf-8"))
        for node in ast.walk(tree):
            targets = []
            if isinstance(node, ast.Assign):
                targets = node.targets
            elif isinstance(node, (ast.AnnAssign, ast.AugAssign)):
                targets = [node.target]
            for t in targets:
                for n in ast.walk(t):
                    if isinstance(n, ast.Name) and n.id in names:
                        found.append(f"{rel}:{node.lineno} defines {n.id}")
            if isinstance(node, ast.ImportFrom):
                for alias in node.names:
                    if alias.name in names:
                        found.append(f"{rel}:{node.lineno} imports {alias.name}")
            if isinstance(node, ast.Attribute) and node.attr in names:
                found.append(f"{rel}:{node.lineno} reads .{node.attr}")
    return found


def test_no_app_module_defines_or_imports_a_removed_global():
    assert offences(FORBIDDEN) == []
```

```python
from typing import Optional

from app.core.calibration import RegimeCal

# Minimum stocks in the average before the reading means anything. The batch
# scorer returns partial results when it hits its deadline, so coverage varies.
MIN_SYMBOLS_FOR_REGIME = 15

# Horizon the association was measured at. Do not relabel this: 21 and 126 days
# were both tested on EGX and neither was significant.
REGIME_HORIZON_DAYS = 63


def classify_regime(mean_score: Optional[float], n_symbols: int,
                    regime: RegimeCal) -> dict:
    """
    Turn the market-wide average composite into a condition reading, using
    the cutoffs and historical record of `regime` (the market's bundle).

    `band` is None when coverage is too thin to average meaningfully; callers
    render "not enough data" rather than a misleading middle reading.
    """
    if mean_score is None or n_symbols < MIN_SYMBOLS_FOR_REGIME:
        return {
            "mean_score": round(mean_score, 1) if mean_score is not None else None,
            "n_symbols": n_symbols,
            "band": None,
            "label": "Not enough data",
            "summary": (
                f"Needs at least {MIN_SYMBOLS_FOR_REGIME} scored stocks to average; "
                f"got {n_symbols}."
            ),
            "horizon_days": REGIME_HORIZON_DAYS,
        }

    if mean_score < regime.weak_max:
        key = "weak"
    elif mean_score < regime.mixed_max:
        key = "mixed"
    else:
        key = "broad"

    band = regime.bands[key]
    return {
        "mean_score": round(mean_score, 1),
        "n_symbols": n_symbols,
        "band": key,
        "label": band.label,
        "summary": band.summary,
        "hist_median_3m_pct": band.median_3m_pct,
        "hist_positive_rate": band.positive_rate,
        "observations": band.observations,
        "horizon_days": REGIME_HORIZON_DAYS,
        # The strength of the association, stated so the UI never implies
        # certainty. `association_significant` False must be honoured.
        "association_rho": regime.rho,
        "association_t": regime.t,
        "association_n": regime.n,
        "association_significant": regime.significant,
        "association_phase_rhos": list(regime.phase_rhos),
    }
```

```python
def _pe_reason(i: int, pe: float, fund: FundamentalsCal, calibrated: bool) -> str:
    median = f"the {fund.label} median"
    if i == 0:
        # Not a bargain — earnings non-recurring or the price collapsed.
        return f"P/E {pe:.1f} — implausibly low; check the earnings are recurring"
    if i == 1:

    if i == 3:
        return f"Above {median} (P/E {pe:.1f})" if calibrated else f"P/E {pe:.1f}"
    if i == 4:
        return f"Expensive on earnings (P/E {pe:.1f})"
    return f"Very expensive (P/E {pe:.1f}) — needs confirmed growth to justify"


def _dy_reason(i: int, dy: float, fund: FundamentalsCal, calibrated: bool) -> str:
```

```python
import json

def make_frame(n: int, seed: int, start: str = "2020-01-01", price: float = 50.0,
               volume: float = 2_000_000.0, vol_spread: float = 0.3) -> pd.DataFrame:
    rng = np.random.default_rng(seed)
    idx = pd.bdate_range(start, periods=n)
    rets = rng.normal(0.0003, 0.012 + 0.003 * (seed % 5), n)
    close = price * np.exp(np.cumsum(rets))
    vols = volume * np.exp(rng.normal(0.0, vol_spread, n))
    return pd.DataFrame({"open": close, "high": close * 1.01, "low": close * 0.99,
                         "close": close, "volume": vols}, index=idx)


def write_frames(cache_dir: str, frames: dict) -> None:
    os.makedirs(cache_dir, exist_ok=True)
    for name, df in frames.items():
        with open(os.path.join(cache_dir, f"{name}.pkl"), "wb") as f:
            pickle.dump(df, f)


def write_json(path: str, value) -> None:
    with open(path, "w", encoding="utf-8") as f:
```

```python
REPO_ROOT = os.path.dirname(SCRIPTS_DIR)

    c = (code or "EG").strip().upper()

                         ("EGX30", "EGX70"), "EGX30 and EGX70")

    """[{symbol, exchange, indices}] — EG from the ticker file, else the cached universe.json."""
    if spec.code == "EG":
        with open(os.path.join(REPO_ROOT, "data", "egx_tickers.json"), encoding="utf-8") as f:
            return [{"symbol": t["symbol"].upper(), "exchange": "EGX",
                     "indices": [t["index"].upper()] if t.get("index") else []}
                    for t in json.load(f)]
    with open(os.path.join(spec.cache_dir, UNIVERSE_FILE), encoding="utf-8") as f:
```

```python
def round_down_nice(x: float) -> float:
    """Largest m*10^k <= x with m in {1, 2, 5}: 23.4M -> 20M, 9.8M -> 5M."""
    if not (isinstance(x, (int, float)) and math.isfinite(x) and x > 0):
        raise CalibrationRefused(f"cannot round {x!r} to a floor")
    k = math.floor(math.log10(x))
    for m in (5, 2, 1):
        if m * 10 ** k <= x:
            return float(m * 10 ** k)
    return float(10 ** k)


def read_pe_rows(db, market_code: str) -> tuple:
    """Current pe_data snapshot: (positive P/Es, positive yields). Fundamentals
    have no history, so this is a snapshot, not a backtest."""
    rows = db.execute("SELECT pe_ratio, dividend_yield FROM pe_data WHERE market = %s",
                      (market_code,)).fetchall()

    def clean(i):
        out = []
        for r in rows:
            v = r[i]
            if v is not None and math.isfinite(float(v)) and float(v) > 0:
                out.append(float(v))
        return out

    return clean(0), clean(1)


def _map_edges(ref_values, ref_edges, target_values, ndigits: int) -> list:
    """Each reference edge's quantile in the reference distribution, read off
    the target distribution: same position, other market."""
    ref = np.asarray(ref_values, dtype=float)
    tgt = np.asarray(target_values, dtype=float)
    out = [round(float(np.quantile(tgt, float((ref < e).mean()))), ndigits) for e in ref_edges]
    if any(b <= a for a, b in zip(out, out[1:])):
        raise CalibrationRefused(f"fitted band edges are not increasing: {out}")
    return out

        if len(values) < MIN_FUNDAMENTALS_ROWS:
            raise CalibrationRefused(
                f"pe_data has {len(values)} usable {name} rows (< {MIN_FUNDAMENTALS_ROWS}); "
                f"refusing to write fundamentals")
    ref = for_market(EG).fundamentals
    pe_edges = _map_edges(eg_pe, [u for u, _ in ref.pe_bands[:-1]], pe, 1)
    dy_edges = _map_edges(eg_dy, [u for u, _ in ref.dy_bands[:-1]], dy, 2)
    fund = FundamentalsCal(
        label=market.short_name,
        pe_median=round(float(np.median(pe)), 1),
        pe_bands=tuple(zip(pe_edges + [math.inf], [p for _, p in ref.pe_bands])),
        dy_median=round(float(np.median(dy)), 2),
        dy_bands=tuple(zip(dy_edges + [math.inf], [p for _, p in ref.dy_bands])),
        dy_mid_under_risk_free=dy_edges[2] < float(market.risk_free_fallback_pct or 0.0),
    )
    # Why only the boundaries move: the median stock must land in the band the
    # EG median lands in, so "having data" is never itself worth points.
    if (band_index(fund.pe_bands, fund.pe_median) != band_index(ref.pe_bands, ref.pe_median)
            or band_index(fund.dy_bands, fund.dy_median) != band_index(ref.dy_bands, ref.dy_median)):
        raise CalibrationRefused(f"the {market.code} median falls in a different band than EG's: {fund}")
    return fund


def fit_turnover_floor(frames: dict) -> float:
    """5th percentile of 60-day mean dollar turnover, rounded down to a round number."""
    values = []
    for df in frames.values():
        t = float((df["close"].astype(float) * df["volume"].astype(float))
                  .iloc[-LIQUIDITY_WINDOW:].mean())
        if math.isfinite(t) and t > 0:
            values.append(t)
    if len(values) < MIN_FLOOR_SYMBOLS:
        raise CalibrationRefused(f"turnover floor: only {len(values)} symbols")
    return round_down_nice(float(np.percentile(values, TURNOVER_FLOOR_PCTILE)))


def volume_samples(frames: dict, symbols) -> np.ndarray:
    """20-day mean share volume, sampled every 21 bars over the trailing year."""
    out = []
    for s in symbols:
        df = frames.get(s)
        if df is None or len(df) < 40:
            continue
        avg = df["volume"].astype(float).iloc[-252:].rolling(20).mean()
        out.extend(v for v in avg.iloc[19::21] if math.isfinite(v))
    return np.asarray(out, dtype=float)


def fit_liquidity_tiers(eg_frames: dict, frames: dict, members, market) -> tuple:
    """
    Index members get the share-volume floors at the same percentiles of THEIR
    distribution that EG's EGX30 floors sit at in the EGX30's. There is no
    history for non-members (the panel is index members only), so other
    listings get those floors scaled by EG's NILEX/EGX30 ratio — the same
    relative leniency NILEX gets against EGX30.
    """
    eg = for_market(EG).liquidity_tiers
    ref = volume_samples(eg_frames, symbols_in_index("EGX30"))
    tgt = volume_samples(frames, members)
    if len(ref) < 30 or len(tgt) < 30:
        raise CalibrationRefused(
            f"liquidity tiers: {len(ref)} EG / {len(tgt)} {market.code} volume samples "
            f"(run `python -m scripts.backtest --market EG --fetch-only` for the EG reference)")
    thin_ref, low_ref = eg["EGX30"]
    p_thin, p_low = float((ref < thin_ref).mean()), float((ref < low_ref).mean())
    thin = int(round_down_nice(float(np.quantile(tgt, max(p_thin, 1e-6)))))
    low = int(round_down_nice(float(np.quantile(tgt, p_low))))
    if low <= thin:
        raise CalibrationRefused(f"liquidity tiers collapsed: thin {thin} >= low {low}")
    n_thin, n_low = eg["NILEX"]
    other = (int(round_down_nice(thin * n_thin / thin_ref)),
             int(round_down_nice(low * n_low / low_ref)))
    tiers = {index_member_tier(market): (thin, low), market.code: other}
    report = {"p_thin": round(p_thin, 3), "p_low": round(p_low, 3),
              "thin_share": round(float((tgt < thin).mean()), 3),
              "n_ref": int(len(ref)), "n_target": int(len(tgt))}
    return tiers, report

def risk_cal_from(risk: dict, fitted_at: str) -> RiskCal:
    if "error" in risk:
        raise CalibrationRefused(f"risk: {risk['error']}")
    keys = ("vol_predicts_vol", "vol_predicts_vol_nonoverlapping",
            "vol_predicts_drawdown", "vol_predicts_drawdown_nonoverlapping")
    for key in keys:
        if risk[key]["ic"] is None:
            raise CalibrationRefused(f"risk: {key} has too few dates for an IC")
    q = {int(r["quintile"]): {"future_vol_ann_pct": r["future_vol_ann_pct"],
                              "median_max_drawdown_pct": r["median_max_drawdown_pct"],
                              "p90_max_drawdown_pct": r["p90_max_drawdown_pct"]}
         for r in risk["quintiles"]}
    if sorted(q) != [1, 2, 3, 4, 5]:
        raise CalibrationRefused(f"risk: quintiles {sorted(q)}")
    v, vn, d, dn = (risk[k] for k in keys)
    return RiskCal(
        fitted_at=fitted_at, n_observations=int(risk["n_observations"]),
        n_symbols=int(risk["n_symbols"]), n_dates=int(risk["n_dates"]),
        lookback_days=int(risk["lookback_days"]), forward_days=int(risk["forward_days"]),
        vol_predicts_vol_ic=v["ic"], vol_predicts_vol_t=v["t"],
        vol_predicts_vol_ic_nonoverlapping=vn["ic"], vol_predicts_vol_t_nonoverlapping=vn["t"],
        vol_predicts_drawdown_ic=d["ic"], vol_predicts_drawdown_t=d["t"],
        vol_predicts_drawdown_ic_nonoverlapping=dn["ic"],
        vol_predicts_drawdown_t_nonoverlapping=dn["t"],
        quintiles=q,
    )


def build_bundle(market, spec: CacheSpec, *, eg_spec: CacheSpec, eg_pe, eg_dy, pe, dy,
                 today: date) -> tuple:
    fitted_at = today.isoformat()
    frames = load_frames(spec, market)
    if not frames:
        raise CalibrationRefused(f"no cached frames in {spec.cache_dir}")
    members = sorted({r["symbol"].upper() for r in read_universe_rows(spec) if r.get("indices")})
    floor = fit_turnover_floor({s: frames[s] for s in members if s in frames})
    price_floor = PRICE_FLOOR[market.code]
    tiers, tier_report = fit_liquidity_tiers(load_frames(eg_spec, EG), frames, members, market)
    fund = fit_fundamentals(eg_pe, eg_dy, pe, dy, market)
    fc_t = forecast_calibration(spec, market, "trailing")
    fc_e = forecast_calibration(spec, market, "ewma")
    reg = regime_calibration(spec)
    risk = risk_calibration(spec, market, floor, price_floor)
    bundle = Calibration(
        market=market.code,
        forecast=forecast_cal_from(fc_t, fc_e, market, fitted_at),
        regime=regime_cal_from(reg, spec, fitted_at),
        risk=risk_cal_from(risk, fitted_at),
        fundamentals=fund,
        liquidity_tiers=tiers,
        liquidity_default_tier=market.code,
        liquid_min_turnover=floor,
        liquid_min_price=price_floor,
        measured_failures=(),
        breadth_measured=False,
        dividend_drag_measured=False,
        provisional=False,
    )
    return bundle, {"regime": reg, "forecast": fc_t, "forecast_ewma": fc_e,
                    "risk": risk, "liquidity": tier_report}


def _py(value, indent: int = 0) -> str:
    """A Python literal for a bundle value; dataclasses become constructor calls."""
    pad, end = "    " * (indent + 1), "    " * indent
    if dataclasses.is_dataclass(value):
        parts = [f"{pad}{f.name}={_py(getattr(value, f.name), indent + 1)},"
                 for f in dataclasses.fields(value)]
        return f"{type(value).__name__}(\n" + "\n".join(parts) + f"\n{end})"
    if isinstance(value, dict):
        if not value:
            return "{}"
        parts = [f"{pad}{_py(k, indent + 1)}: {_py(v, indent + 1)}," for k, v in value.items()]
        return "{\n" + "\n".join(parts) + f"\n{end}}}"
    if isinstance(value, tuple):
        if not value:
            return "()"
        return "(" + ", ".join(_py(v, indent + 1) for v in value) + ",)"
    if isinstance(value, bool) or value is None:
        return repr(value)
    if isinstance(value, float):
        if math.isinf(value):
            return "math.inf" if value > 0 else "-math.inf"
        return repr(value)
    if isinstance(value, (int, str)):
        return repr(value)
    raise TypeError(f"cannot render {type(value).__name__}")
```
